giờ drain từng pg theo batch 1 
for BATCH in "$EVIDENCE"/batch-*; do
    echo
    echo "========================================="
    echo "Running: $BATCH"
    echo "========================================="

    while IFS= read -r PG; do
        [ -z "$PG" ] && continue

        echo "UPMAP $PG : osd.$X -> osd.$S"

        ceph osd pg-upmap-items "$PG" "$X" "$S" </dev/null || exit 1
    done < "$BATCH"

    echo "Waiting for all PGs to become active+clean..."

    while :; do
        BAD=$(
            ceph pg dump pgs_brief -f json 2>/dev/null \
            | jq '[.pg_stats[] | select(.state != "active+clean")] | length'
        )

        echo "PG not active+clean: $BAD"

        if [ "$BAD" -eq 0 ]; then
            echo "Batch converged."
            break
        fi

        sleep 5
    done
done
